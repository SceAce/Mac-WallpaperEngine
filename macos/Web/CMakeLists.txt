cmake_minimum_required(VERSION 3.24)
project(VividWeb LANGUAGES C CXX OBJCXX)
set(CMAKE_OSX_ARCHITECTURES arm64)
set(PROJECT_ARCH arm64)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CEF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.build/dependencies/cef-download/cef_binary_144.0.35+g12f1636+chromium-144.0.7559.262_macosarm64_minimal" CACHE PATH "Pinned CEF distribution")
set(USE_SANDBOX OFF CACHE BOOL "Development CEF packaging")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0")
add_executable(VividWebService MACOSX_BUNDLE Service.mm)
set_target_properties(VividWebService PROPERTIES BUNDLE_EXTENSION xpc MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Service.plist")
add_executable(VividWebHelper Helper.mm)
foreach(target VividWebService VividWebHelper)
    target_include_directories(${target} SYSTEM PRIVATE "${CEF_ROOT}")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(${target} PRIVATE libcef_dll_wrapper "-framework AppKit" "-framework Metal" "-framework IOSurface" "-framework CoreVideo")
    target_compile_options(${target} PRIVATE -fobjc-arc -Wall -Wextra -Werror -Wno-unused-parameter)
endforeach()
