cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum macOS version")
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
project(VividScene LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "Vivid scene requires Apple Silicon macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(VIVID_MOLTENVK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.build/dependencies/MoltenVK/MoltenVK"
    CACHE PATH "MoltenVK 1.4.2 distribution")
set(Vulkan_INCLUDE_DIR "${VIVID_MOLTENVK_ROOT}/include" CACHE PATH "Vulkan headers")
set(Vulkan_LIBRARY "${VIVID_MOLTENVK_ROOT}/dynamic/dylib/macOS/libMoltenVK.dylib"
    CACHE FILEPATH "Vulkan implementation")
set(HANABI_BUILD_VENDORED_DXC OFF CACHE BOOL "Use the separately built pinned DXC")
set(HANABI_DXC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.build/dxc-stage" CACHE PATH "DXC SDK")
set(BUILD_QML OFF CACHE BOOL "No Qt host on macOS")
set(CMAKE_BUILD_RPATH "${VIVID_MOLTENVK_ROOT}/dynamic/dylib/macOS;${HANABI_DXC_ROOT}/lib")
add_subdirectory("../../producer/third_party/wallpaper-scene-renderer" renderer)
add_executable(VividSceneService MACOSX_BUNDLE Service.mm Project.cpp)
set_target_properties(VividSceneService PROPERTIES BUNDLE_EXTENSION xpc
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Service.plist")
target_link_libraries(VividSceneService PRIVATE wescene-renderer wpVulkan nlohmann_json
    "-framework Foundation" "-framework IOSurface")
target_compile_options(VividSceneService PRIVATE -Wall -Wextra -Werror
    "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
if(BUILD_TESTING)
    add_executable(scene_project_test Tests/ProjectTest.cpp Project.cpp)
    target_include_directories(scene_project_test PRIVATE .)
    target_link_libraries(scene_project_test PRIVATE wescene-renderer nlohmann_json)
    target_compile_options(scene_project_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME scene_project COMMAND scene_project_test)
endif()
